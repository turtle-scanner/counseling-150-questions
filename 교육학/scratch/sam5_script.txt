    // Web Audio 신디사이저

    let audioCtx = null;
    function getAudioContext() {
      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      return audioCtx;
    }

    function playSound(type) {
      if (!gameState.soundEnabled) return;
      try {
        const actx = getAudioContext();
        if (actx.state === 'suspended') actx.resume();

        const now = actx.currentTime;
        const osc = actx.createOscillator();
        const gain = actx.createGain();
        osc.connect(gain);
        gain.connect(actx.destination);

        if (type === 'slash') {
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(500, now);
          osc.frequency.exponentialRampToValueAtTime(70, now + 0.16);
          gain.gain.setValueAtTime(0.45, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.16);
          osc.start(now);
          osc.stop(now + 0.16);
        } else if (type === 'arrow') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(900, now);
          osc.frequency.exponentialRampToValueAtTime(280, now + 0.2);
          gain.gain.setValueAtTime(0.3, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
          osc.start(now);
          osc.stop(now + 0.2);
        } else if (type === 'fire') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(140, now);
          osc.frequency.exponentialRampToValueAtTime(25, now + 0.4);
          gain.gain.setValueAtTime(0.55, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
          osc.start(now);
          osc.stop(now + 0.4);
        } else if (type === 'fanfare') {
          osc.type = 'square';
          osc.frequency.setValueAtTime(349, now);
          osc.frequency.setValueAtTime(440, now + 0.09);
          osc.frequency.setValueAtTime(523, now + 0.18);
          gain.gain.setValueAtTime(0.25, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
          osc.start(now);
          osc.stop(now + 0.35);
        } else if (type === 'drum') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(120, now);
          osc.frequency.exponentialRampToValueAtTime(30, now + 0.25);
          gain.gain.setValueAtTime(0.6, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
          osc.start(now);
          osc.stop(now + 0.25);
        }
      } catch (e) {
        console.warn("Audio error:", e);
      }
    }

    // ==========================================================================

    function getHexPixel(q, r) {
      const x = 52 + q * HEX_W + (r % 2 === 1 ? HEX_W / 2 : 0);
      const y = 46 + r * (HEX_H * 0.75);
      return { x, y };
    }

    function getHexDistance(q1, r1, q2, r2) {
      const x1 = q1 - (r1 - (r1 & 1)) / 2;
      const z1 = r1;
      const y1 = -x1 - z1;
      const x2 = q2 - (r2 - (r2 & 1)) / 2;
      const z2 = r2;
      const y2 = -x2 - z2;
      return Math.max(Math.abs(x1 - x2), Math.abs(y1 - y2), Math.abs(z1 - z2));
    }

    function findUnitAt(q, r) {
      return UNITS.find(u => u.q === q && u.r === r && u.troops > 0);
    }

    // ==========================================================================
    // 캔버스 렌더링 시스템 (불길 확산 & 헥스 맵)
    // ==========================================================================
    function drawHexagon(ctx, x, y, radius, fillColor, strokeColor, lineWidth = 1) {
      ctx.beginPath();
      for (let i = 0; i < 6; i++) {
        const angle = (Math.PI / 180) * (60 * i - 30);
        const hx = x + radius * Math.cos(angle);
        const hy = y + radius * Math.sin(angle);
        if (i === 0) ctx.moveTo(hx, hy);
        else ctx.lineTo(hx, hy);
      }
      ctx.closePath();
      if (fillColor) {
        ctx.fillStyle = fillColor;
        ctx.fill();
      }
      if (strokeColor) {
        ctx.strokeStyle = strokeColor;
        ctx.lineWidth = lineWidth;
        ctx.stroke();
      }
    }

    function renderBattlefield() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // 1. 헥스 타일 렌더링
      for (let r = 0; r < GRID_ROWS; r++) {
        for (let q = 0; q < GRID_COLS; q++) {
          const { x, y } = getHexPixel(q, r);
          const terrain = MAP_TERRAIN[r][q];

          let tileColor = "#162032";
          let borderColor = "#334155";

          if (terrain === 1) tileColor = "#14532d"; // 숲
          else if (terrain === 2) tileColor = "#44403c"; // 산악
          else if (terrain === 3) tileColor = "#7f1d1d"; // 호로관
          else if (terrain === 4) tileColor = gameState.supplyBurned ? "#292524" : "#854d0e"; // 보급창

          // 이동/공격 하이라이트
          const isReachable = gameState.reachableHexes.some(h => h.q === q && h.r === r);
          if (isReachable) {
            tileColor = "#065f46";
            borderColor = "#34d399";
          }

          const isAttackable = gameState.attackableTargets.some(t => t.q === q && t.r === r);
          if (isAttackable) {
            tileColor = "#991b1b";
            borderColor = "#f87171";
          }

          const isBurning = gameState.fireTiles.some(f => f.q === q && f.r === r);
          if (isBurning) {
            tileColor = "#c2410c";
          }

          drawHexagon(ctx, x, y, HEX_RADIUS - 1, tileColor, borderColor, isReachable || isAttackable ? 2.5 : 1);

          // 지형 텍스트 라벨
          ctx.font = "bold 10px sans-serif";
          ctx.fillStyle = "rgba(255,255,255,0.45)";
          ctx.textAlign = "center";
          if (terrain === 1) ctx.fillText("🌲숲", x, y - 16);
          else if (terrain === 2) ctx.fillText("⛰️산", x, y - 16);
          else if (terrain === 3) ctx.fillText("🏯호로관", x, y - 16);
          else if (terrain === 4) ctx.fillText(gameState.supplyBurned ? "💥소실" : "🌾군량고", x, y - 16);

          if (isBurning) {
            ctx.font = "15px sans-serif";
            ctx.fillText("🔥", x, y + 18);
          }
        }
      }

      // 2. 부대 렌더링
      UNITS.forEach(unit => {
        if (unit.troops <= 0) return;
        const { x, y } = getHexPixel(unit.q, unit.r);
        const isSelected = gameState.selectedUnit && gameState.selectedUnit.id === unit.id;
        const isAlly = unit.faction === "ally";

        // 선택 하이라이트 원
        if (isSelected) {
          ctx.beginPath();
          ctx.arc(x, y, HEX_RADIUS - 4, 0, Math.PI * 2);
          ctx.strokeStyle = "#facc15";
          ctx.lineWidth = 3.5;
          ctx.stroke();
        }

        // 베이스 원형
        ctx.beginPath();
        ctx.arc(x, y, 22, 0, Math.PI * 2);
        ctx.fillStyle = isAlly ? "#1d4ed8" : (unit.isBoss ? "#7f1d1d" : "#b91c1c");
        ctx.fill();
        ctx.strokeStyle = unit.acted ? "#475569" : (isAlly ? "#93c5fd" : "#fca5a5");
        ctx.lineWidth = unit.isBoss ? 3 : 2;
        ctx.stroke();

        // 행동 완료 표시 음영
        if (unit.acted) {
          ctx.fillStyle = "rgba(0,0,0,0.5)";
          ctx.beginPath();
          ctx.arc(x, y, 22, 0, Math.PI * 2);
          ctx.fill();
        }

        // 아바타
        ctx.font = "18px serif";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
        ctx.fillText(unit.avatar, x, y - 2);

        // 무장명
        ctx.font = "bold 11px sans-serif";
        ctx.fillStyle = unit.isBoss ? "#fef08a" : "#ffffff";
        ctx.shadowColor = "#000000";
        ctx.shadowBlur = 4;
        ctx.fillText(unit.name, x, y + 14);
        ctx.shadowBlur = 0;

        // 삼국지 5 진형 배지 표시
        const formObj = FORMATIONS[unit.formation];
        if (formObj) {
          ctx.font = "bold 9px sans-serif";
          ctx.fillStyle = formObj.color;
          ctx.fillText(`[${formObj.name.substring(0, 2)}]`, x, y - 18);
        }

        // 병력 게이지
        const hpPercent = Math.max(0, unit.troops / unit.maxTroops);
        const barW = 34;
        const barH = 4;
        ctx.fillStyle = "#0f172a";
        ctx.fillRect(x - barW / 2, y + 23, barW, barH);
        ctx.fillStyle = isAlly ? (hpPercent > 0.4 ? "#22c55e" : "#ef4444") : (unit.isBoss ? "#e11d48" : "#f97316");
        ctx.fillRect(x - barW / 2, y + 23, barW * hpPercent, barH);
      });
    }

    // ==========================================================================
    // 마우스 상호작용
    // ==========================================================================
    canvas.addEventListener("click", function(e) {
      if (gameState.battleOver) return;
      const rect = canvas.getBoundingClientRect();
      const scaleX = canvas.width / rect.width;
      const scaleY = canvas.height / rect.height;
      const mx = (e.clientX - rect.left) * scaleX;
      const my = (e.clientY - rect.top) * scaleY;

      let closestHex = null;
      let minDist = 9999;
      for (let r = 0; r < GRID_ROWS; r++) {
        for (let q = 0; q < GRID_COLS; q++) {
          const { x, y } = getHexPixel(q, r);
          const dist = Math.hypot(mx - x, my - y);
          if (dist < minDist && dist < HEX_RADIUS) {
            minDist = dist;
            closestHex = { q, r };
          }
        }
      }

      if (!closestHex) return;
      handleHexClick(closestHex.q, closestHex.r);
    });

    function handleHexClick(q, r) {
      // 1. 공격/특기 발동 모드
      if (gameState.skillMode) {
        const target = gameState.attackableTargets.find(t => t.q === q && t.r === r);
        if (target) {
          executeSkillAction(gameState.skillMode, target);
        } else {
          logBattle("명령 대상이 올바르지 않아 군령을 철회했습니다.");
        }
        clearSkillMode();
        renderBattlefield();
        return;
      }

      // 2. 이동 타일 클릭 시 (이동 실행)
      const canMove = gameState.reachableHexes.find(h => h.q === q && h.r === r);
      if (canMove && gameState.selectedUnit && !gameState.selectedUnit.moved) {
        gameState.selectedUnit.q = q;
        gameState.selectedUnit.r = r;
        gameState.selectedUnit.moved = true;
        gameState.reachableHexes = [];

        // 특수 기믹: 보급창(8, 3) 도달 검사
        if (q === 8 && r === 3 && !gameState.supplyBurned) {
          triggerSupplyRaid(gameState.selectedUnit);
        } else {
          logBattle(`${gameState.selectedUnit.name} 부대가 진격하여 진형을 잡았습니다.`);
        }

        playSound('fanfare');
        updateActionCommands();
        renderBattlefield();
        return;
      }

      // 3. 부대 선택
      const unit = findUnitAt(q, r);
      if (unit) {
        selectUnit(unit);
      } else {
        deselectUnit();
      }
      renderBattlefield();
    }

    function selectUnit(unit) {
      gameState.selectedUnit = unit;
      updateUnitDetailCard(unit);

      if (unit.faction === "ally" && !unit.moved && !unit.acted && gameState.phase === "ally") {
        computeMovementRange(unit);
      } else {
        gameState.reachableHexes = [];
      }

      gameState.attackableTargets = [];
      updateActionCommands();
    }

    function deselectUnit() {
      gameState.selectedUnit = null;
      gameState.reachableHexes = [];
      gameState.attackableTargets = [];
      clearSkillMode();
      document.getElementById("unitName").innerText = "부대를 선택하세요";
      document.getElementById("unitGeneralTitle").innerText = "-";
      document.getElementById("unitTroops").innerText = "-";
      document.getElementById("unitMorale").innerText = "-";
      document.getElementById("statWar").innerText = "-";
      document.getElementById("statInt").innerText = "-";
      document.getElementById("statLead").innerText = "-";
      document.getElementById("statChr").innerText = "-";
      document.getElementById("unitFormationBadge").classList.add("hidden");
      document.getElementById("unitSkillsList").innerHTML = `<span class="text-slate-500">부대를 클릭하십시오</span>`;
      disableAllCommands();
    }

    // ==========================================================================
    // 전략 기믹 1: 보급창고(군량고) 기습 점령 이벤트
    // ==========================================================================
    function triggerSupplyRaid(hero) {
      gameState.supplyBurned = true;
      logBattle(`🌾💥 [전략 성공!] ${hero.name}이 적의 보급창고를 불태웠습니다! 동탁군 전원 패닉 & 사기 급락!`);
      playSound('fire');

      // 적 전군 사기 25로 추락 및 병력 25% 이탈
      UNITS.filter(u => u.faction === "enemy" && u.troops > 0).forEach(enemy => {
        enemy.morale = Math.max(15, enemy.morale - 45);
        const lost = Math.floor(enemy.troops * 0.25);
        enemy.troops -= lost;
        showDamagePop(enemy.q, enemy.r, `군량상실 -${lost}`);
      });

      document.getElementById("tacticalAlertText").innerText = "🎉 적 군량고 소실! 10턴 내에 전멸시킬 수 있는 절호의 기회입니다!";
    }

    // ==========================================================================
    // 이동 범위 연산 (추행진 기동력 +2 적용)
    // ==========================================================================
    function computeMovementRange(unit) {
      gameState.reachableHexes = [];
      const form = FORMATIONS[unit.formation];
      const baseMove = 2 + (form ? form.moveBonus : 0);

      for (let r = 0; r < GRID_ROWS; r++) {
        for (let q = 0; q < GRID_COLS; q++) {
          if (q === unit.q && r === unit.r) continue;
          if (findUnitAt(q, r)) continue;

          const terrain = MAP_TERRAIN[r][q];
          // 호로관(3) 성벽은 직접 진입 불가 (우회 돌파)
          if (terrain === 3 && unit.faction === "ally") continue;

          const dist = getHexDistance(unit.q, unit.r, q, r);
          if (dist <= baseMove) {
            gameState.reachableHexes.push({ q, r });
          }
        }
      }
    }

    function updateActionCommands() {
      const u = gameState.selectedUnit;
      if (!u || u.faction !== "ally" || u.acted || gameState.phase !== "ally") {
        disableAllCommands();
        return;
      }

      const form = FORMATIONS[u.formation];
      const adjEnemies = getEnemiesInRange(u, 1);
      const arcRange = form ? form.arcRange : 1;
      const arcEnemies = getEnemiesInRange(u, arcRange);

      document.getElementById("btnAttack").disabled = adjEnemies.length === 0;
      document.getElementById("btnArchery").disabled = arcEnemies.length === 0 || arcRange < 2;
      document.getElementById("btnFireSkill").disabled = !u.skills.includes("화계") && u.int < 75;
      document.getElementById("btnChaosSkill").disabled = !u.skills.includes("혼란") && u.int < 80;
      document.getElementById("btnFlurry").disabled = !u.skills.includes("분기") || adjEnemies.length === 0;
      document.getElementById("btnFormation").disabled = false;
      document.getElementById("btnDuel").disabled = adjEnemies.length === 0;
      document.getElementById("btnWait").disabled = false;
    }

    function getEnemiesInRange(unit, range) {
      return UNITS.filter(enemy => 
        enemy.faction !== unit.faction &&
        enemy.troops > 0 &&
        getHexDistance(unit.q, unit.r, enemy.q, enemy.r) <= range &&
        getHexDistance(unit.q, unit.r, enemy.q, enemy.r) > 0
      );
    }

    function onCmdFlurry() {
      const u = gameState.selectedUnit;
      if (!u) return;

      if (enemies.length === 0) return;

      logBattle(`⚡ ${u.name}의 [분기(奮起)] 발동! 주변 모든 적군을 일격에 베어냅니다!`);
      playSound('slash');

      enemies.forEach(target => {
        const dmg = calculateCombatDamage(u, target, 'flurry');

    function executeSkillAction(mode, targetPos) {
      const u = gameState.selectedUnit;
      const targetUnit = findUnitAt(targetPos.q, targetPos.r);

      if (mode === 'attack' && targetUnit) {
        logBattle(`🗡️ ${u.name} 부대가 ${targetUnit.name}에게 돌격을 감행합니다!`);
        playSound('slash');
        const dmg = calculateCombatDamage(u, targetUnit, 'melee');
        applyDamage(targetUnit, dmg);

        if (targetUnit.troops > 0) {
          const counterDmg = Math.floor(calculateCombatDamage(targetUnit, u, 'melee') * 0.45);
          applyDamage(u, counterDmg);
        }
      } else if (mode === 'archery' && targetUnit) {
        logBattle(`🏹 ${u.name} 부대의 화살비가 쏟아집니다!`);
        playSound('arrow');
        const dmg = calculateCombatDamage(u, targetUnit, 'archery');
        applyDamage(targetUnit, dmg);
      } else if (mode === 'fire') {
        logBattle(`🔥 [대화계] 불길이 거세게 치솟습니다! 남동풍을 타고 번져갑니다.`);
        playSound('fire');
        gameState.fireTiles.push({ q: targetPos.q, r: targetPos.r, turns: 3 });
        if (targetUnit) {
          applyDamage(targetUnit, 1500);
          targetUnit.morale = Math.max(10, targetUnit.morale - 20);
        }
      } else if (mode === 'chaos' && targetUnit) {
        const successRate = 55 + (u.int - targetUnit.int) * 2;
        if (Math.random() * 100 < successRate) {
          targetUnit.morale = Math.max(10, targetUnit.morale - 30);
          targetUnit.acted = true;
          logBattle(`🌀 [혼란 대성공] ${targetUnit.name}의 군심이 무너져 행동 불능이 되었습니다!`);
          playSound('fanfare');
        } else {
          logBattle(`[혼란 실패] ${targetUnit.name}이 계략을 눈치챘습니다.`);
        }
      }

      finishUnitAction(u);
    }

    // 10턴 결전 전용 고속 밸런스 데미지 공식
    function calculateCombatDamage(attacker, defender, type) {
      const aForm = FORMATIONS[attacker.formation] || FORMATIONS.gyorin;
      const dForm = FORMATIONS[defender.formation] || FORMATIONS.gyorin;

      let atkPower = (attacker.war * 0.65 + attacker.lead * 0.35) * (attacker.troops / 4500);
      let defPower = (defender.lead * 0.65 + defender.war * 0.35) * (defender.troops / 4500);

      if (type === 'archery') {
        atkPower *= aForm.arcMod;
        if (defender.formation === 'bangwon') defPower *= 3.0; // 방원진 활 철벽 방어
      } else {
        atkPower *= aForm.atkMod;
        defPower *= dForm.defMod;
      }

      atkPower *= (attacker.morale / 100);

      // 10턴 내에 승부가 시원하게 나도록 대미지 배율 튜닝 (1.4배)
      let baseDamage = Math.floor(((atkPower / Math.max(10, defPower)) * 1200 + Math.random() * 300) * 1.4);
      return Math.max(400, baseDamage);
    }

    function applyDamage(unit, damage) {
      unit.troops = Math.max(0, unit.troops - damage);
      showDamagePop(unit.q, unit.r, `-${damage}`);

      if (unit.troops <= 0) {
        logBattle(`☠️ [궤멸] 적장 ${unit.name} 부대가 전멸하여 패주했습니다!`);
        checkVictoryCondition();
      }
      renderBattlefield();
    }

    function showDamagePop(q, r, text) {
      const { x, y } = getHexPixel(q, r);
      const container = document.getElementById("damageOverlay");
      const pop = document.createElement("div");
      pop.className = "damage-pop text-red-500 font-black";
      pop.innerText = text;
      pop.style.left = `${x}px`;
      pop.style.top = `${y}px`;
      container.appendChild(pop);
      setTimeout(() => pop.remove(), 850);
    }

    function progressDuelRound() {
      const { p1, p2 } = duelState;
      duelState.round++;
      playSound('slash');

      const p1Roll = p1.war + Math.floor(Math.random() * 25);
      const p2Roll = p2.war + Math.floor(Math.random() * 25);

      const logBox = document.getElementById("duelNarrative");
      const vsIcon = document.getElementById("duelVsIcon");
      vsIcon.classList.add("clash-anim");
      setTimeout(() => vsIcon.classList.remove("clash-anim"), 400);

      if (p1Roll >= p2Roll) {
        const dmg = Math.floor((p1Roll - p2Roll) * 1.8) + 20;
        duelState.p2Hp = Math.max(0, duelState.p2Hp - dmg);
        logBox.innerText = `[제${duelState.round}합] ${p1.name}의 일격필살! ${p2.name}에게 ${dmg}의 타격을 입혔습니다!`;
      } else {
        const dmg = Math.floor((p2Roll - p1Roll) * 1.8) + 20;
        duelState.p1Hp = Math.max(0, duelState.p1Hp - dmg);
        logBox.innerText = `[제${duelState.round}합] ${p2.name}의 거친 방천화극 반격! ${p1.name}이 부상을 입었습니다 (-${dmg})!`;
      }

      document.getElementById("duelP1Hp").style.width = `${duelState.p1Hp}%`;
      document.getElementById("duelP2Hp").style.width = `${duelState.p2Hp}%`;

      // 승패 판정
      if (duelState.p1Hp <= 0 || duelState.p2Hp <= 0 || duelState.round >= 5) {
        document.getElementById("btnDuelRound").classList.add("hidden");
        document.getElementById("btnDuelClose").classList.remove("hidden");

        if (duelState.p2Hp <= 0) {
          logBox.innerText = `🏆 [초대형 전공!] ${p1.name}이 적장 ${p2.name}을 베어 낙마시켰습니다! 적 군세가 완전히 와해되었습니다!`;
          applyDamage(p2, p2.troops); // 적장 전멸
          if (p2.isBoss) {
            // 여포 격파 시 즉시 승리!
            setTimeout(() => {
              triggerVictory("일기토 승리! 적 총대장 여포를 꺾고 호로관을 함락시켰습니다!");
            }, 600);
          }
        } else if (duelState.p1Hp <= 0) {
          logBox.innerText = `💀 ${p1.name}이 패배하여 전선에서 후퇴합니다...`;
          applyDamage(p1, p1.troops);
        } else {
          logBox.innerText = `🤝 5합 동안 결판을 내지 못하고 서로 물러섰습니다.`;
        }
      }
    }

    function endPlayerTurn() {
      if (gameState.phase !== "ally" || gameState.battleOver) return;
      gameState.phase = "enemy";
      document.getElementById("btnEndTurn").disabled = true;
      document.getElementById("turnInfo").innerText = `제 ${gameState.turn} / 10 턴 [동탁군 반격 개시]`;
      logBattle("🚩 동탁군이 북을 울리며 전면 반격을 개시합니다!");
      playSound('drum');

      setTimeout(() => {
        runEnemyAI();
      }, 700);
    }

    function runEnemyAI() {
      const enemies = UNITS.filter(u => u.faction === "enemy" && u.troops > 0);
      let step = 0;

      function executeNextEnemy() {
        if (step >= enemies.length || gameState.battleOver) {
          finishFullTurn();
          return;
        }

        const enemy = enemies[step];
        const allies = UNITS.filter(u => u.faction === "ally" && u.troops > 0);

        if (allies.length > 0) {
          let closestAlly = null;
          let minDist = 999;
          allies.forEach(a => {
            const d = getHexDistance(enemy.q, enemy.r, a.q, a.r);
            if (d < minDist) {
              minDist = d;
              closestAlly = a;
            }
          });

          if (minDist === 1) {
            logBattle(`⚔️ 적장 ${enemy.name}이 ${closestAlly.name}에게 맹공을 퍼붓습니다!`);
            playSound('slash');
            const dmg = calculateCombatDamage(enemy, closestAlly, 'melee');
            applyDamage(closestAlly, dmg);
          } else {
            // 아군을 향해 진격
            const stepQ = enemy.q > closestAlly.q ? enemy.q - 1 : (enemy.q < closestAlly.q ? enemy.q + 1 : enemy.q);
            const stepR = enemy.r > closestAlly.r ? enemy.r - 1 : (enemy.r < closestAlly.r ? enemy.r + 1 : enemy.r);
            if (!findUnitAt(stepQ, stepR)) {
              enemy.q = stepQ;
              enemy.r = stepR;
              logBattle(`🏇 적군 ${enemy.name} 부대가 진형을 좁히며 쇄도합니다.`);
            }
          }
        }

        enemy.acted = true;
        renderBattlefield();
        step++;
        setTimeout(executeNextEnemy, 550);
      }

      executeNextEnemy();
    }

    // 턴 교대 및 화계 번짐 연산
    function finishFullTurn() {
      gameState.turn++;

      // 10턴 초과 시 타임오버 패배 판정!
      if (gameState.turn > MAX_TURNS) {
        triggerDefeat("10턴이 경과하여 동탁의 서량 기병 10만 본대가 당도했습니다! 연합군은 퇴로가 끊겨 패주했습니다.");
        return;
      }

      gameState.phase = "ally";
      document.getElementById("btnEndTurn").disabled = false;
      document.getElementById("turnInfo").innerText = `제 ${gameState.turn} / 10 턴 [연합군 행동]`;

      // 10턴 게이지 업데이트
      const turnsLeft = MAX_TURNS - gameState.turn + 1;
      document.getElementById("turnCountdownText").innerText = `${turnsLeft} 턴 남음!`;
      const percent = (turnsLeft / MAX_TURNS) * 100;
      const barEl = document.getElementById("turnProgressBar");
      barEl.style.width = `${percent}%`;

      if (turnsLeft <= 3) {
        barEl.className = "turn-meter-bar h-full rounded-full bg-red-600 animate-pulse";
        document.getElementById("turnCountdownText").className = "text-red-500 font-black animate-ping";
      }

      // 전략 기믹 3: 남동풍을 타고 화계(불길) 확산
      const newFires = [];
      gameState.fireTiles.forEach(f => {
        f.turns--;
        if (f.turns > 0) {
          newFires.push(f);
          // 40% 확률로 남동쪽 (q+1, r) 또는 (q, r+1)로 불길 확산
          if (Math.random() < 0.45 && f.q + 1 < GRID_COLS) {
            newFires.push({ q: f.q + 1, r: f.r, turns: 2 });
          }
        }
      });
      gameState.fireTiles = newFires;

      // 불타는 타일 위의 부대에게 화염 피해
      gameState.fireTiles.forEach(f => {
        const u = findUnitAt(f.q, f.r);
        if (u) {
          applyDamage(u, 750);
          logBattle(`🔥 ${u.name} 부대가 불길에 휩싸여 큰 피해를 입었습니다!`);
        }
      });

      // 부대 턴 리셋
      UNITS.forEach(u => {
        u.moved = false;
        u.acted = false;
      });

      logBattle(`제 ${gameState.turn} 턴 시작! 남은 기회는 ${turnsLeft}번입니다.`);
      deselectUnit();
      renderBattlefield();
    }

    // ==========================================================================
    // 승패 판정 시스템 (10턴 룰)
    // ==========================================================================
    function checkVictoryCondition() {
      const enemiesAlive = UNITS.some(u => u.faction === "enemy" && u.troops > 0);
      const alliesAlive = UNITS.some(u => u.faction === "ally" && u.troops > 0);
      const boss = UNITS.find(u => u.isBoss);

      if (boss && boss.troops <= 0) {
        triggerVictory("동탁군의 총대장 여포가 퇴각했습니다! 적 군세가 붕괴하여 대승을 거두었습니다!");
      } else if (!enemiesAlive) {
        triggerVictory("동탁군 전 부대를 궤멸시켰습니다! 천하에 이름을 떨친 완벽한 승리입니다!");
      } else if (!alliesAlive) {
        triggerDefeat("연합군 전 부대가 전멸하여 패배했습니다...");
      }
    }

    function triggerVictory(msg) {
      if (gameState.battleOver) return;
      gameState.battleOver = true;
      playSound('fanfare');
      alert(`🎉 [대승리 - 10턴 결전 완료]\n\n${msg}\n소요 턴: ${gameState.turn}턴만에 완벽한 승리를 거두었습니다!`);
      logBattle(`🏆 대승리! ${msg}`);
    }

    function triggerDefeat(msg) {
      if (gameState.battleOver) return;
      gameState.battleOver = true;
      playSound('fire');
      alert(`💀 [작전 실패 - 시간 초과 패배]\n\n${msg}`);
      logBattle(`통한의 패배: ${msg}`);
    }

    function restartBattle() {
      location.reload();
    }

    function logBattle(msg) {
      const logEl = document.getElementById("battleLog");
      logEl.innerText = msg;
    }
